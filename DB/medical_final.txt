-- Medical Table

CREATE TABLE MEDICAL (
    Medical_id INT AUTO_INCREMENT PRIMARY KEY,
    Ug_id CHAR(6),
    From_date DATE,
    To_date DATE,
    Status ENUM('Pending','MedicalApproved','MedicalDeclined'),
    Session_type ENUM('Lecture','Exam'),
    Techoff_id CHAR(6),
    C_code CHAR(7),

    FOREIGN KEY (Ug_id) REFERENCES UNDERGRADUATE(Ug_id) ON DELETE CASCADE,
    FOREIGN KEY (Techoff_id) REFERENCES TECH_OFFICER(Techoff_id) ON DELETE SET NULL,
    FOREIGN KEY (C_code) REFERENCES COURSE(C_code) ON DELETE CASCADE
);
