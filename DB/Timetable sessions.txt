--Timetable (Sessions)
CREATE TABLE Timetable(
    Timetable_id CHAR(6),
    Day ENUM('Mon','Tue','Wed','Thu','Fri','Sat','Sun'),
    Start_time TIME,
    End_time TIME,
    C_code CHAR(8),
    Type ENUM('T','P'),
    PRIMARY KEY (Timetable_id),
    FOREIGN KEY (C_code) REFERENCES COURSE(C_code) ON DELETE CASCADE
);
