DROP DATABASE IF EXISTS LMS_DB;
CREATE DATABASE LMS_DB;
USE LMS_DB;

-- User
CREATE TABLE USER (
    User_id CHAR(6) PRIMARY KEY,
    Password VARCHAR(100),
    Role ENUM('Admin','Lecturer','Tech_Officer','Undergraduate'),
    Profile_pic VARCHAR(255)
);


-- ADMIN
CREATE TABLE ADMIN (
    Admin_id CHAR(6) PRIMARY KEY,
    Name VARCHAR(100),
    Email VARCHAR(100),
    Nic VARCHAR(12),
    Dob DATE,
    Dpt_name VARCHAR(50),
    Phone VARCHAR(15),
    No VARCHAR(10),
    Street VARCHAR(100),
    City VARCHAR(50),

    FOREIGN KEY (Admin_id) REFERENCES USER(User_id) ON DELETE CASCADE
);

-- LECTURER
CREATE TABLE LECTURER (
    Lec_id CHAR(6) PRIMARY KEY,
    Name VARCHAR(100),
    Email VARCHAR(100),
    Nic VARCHAR(12),
    Dob DATE,
    Education_lvl ENUM('Bachelor','Master','Phd'),
    Dpt_name VARCHAR(50),
    No VARCHAR(10),
    Street VARCHAR(100),
    City VARCHAR(50),

    FOREIGN KEY (Lec_id) REFERENCES USER(User_id)
    ON DELETE CASCADE
);

CREATE TABLE LECTURER_PHONE (
    Lec_id CHAR(6),
    Phone CHAR(10),

    PRIMARY KEY (Lec_id, Phone),
    FOREIGN KEY (Lec_id) REFERENCES LECTURER(Lec_id)
);
